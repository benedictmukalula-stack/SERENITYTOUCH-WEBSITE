    31→    31→  try {
    32→    32→    const now = Date.now();
    33→    33→    const then = new Date(dateStr).getTime();
    34→    34→    const diff = Math.floor((now - then) / 1000);
    35→    35→    if (diff < 60) return `${diff}s ago`;
    36→    36→    if (diff < 3600) return `${Math.floor(diff / 60)} min ago`;
    37→    37→    if (diff < 86400) return `${Math.floor(diff / 3600)} hour ago`;
    38→    38→    return `${Math.floor(diff / 86400)} days ago`;
    39→    39→  } catch {
    40→    40→    return '';
