    64→function getServiceLabel(val: string) { return serviceOptions.find(s => s.value === val)?.label || val; }
    65→function getTherapistLabel(val: string) { return therapistOptions.find(t => t.value === val)?.label || val; }

    68→function getCalloutLabel(val: string) {
    69→  const z = calloutZones.find(z => z.id === val);
    70→  return z ? (z.fee > 0 ? `${z.label} (K${z.fee} call-out fee)` : `${z.label} (Custom quote)`) : val;

    73→function buildWhatsAppMessage(data: typeof defaultFormData): string {
    74→  const lines = [
    75→    `*Tina's Sanctuary — Booking Request*`, ``,
    76→    `*Name:* ${data.name}`, `*Email:* ${data.email}`, `*Phone:* ${data.phone || 'Not provided'}`,
    77→    `*Date:* ${data.date}`, `*Service:* ${getServiceLabel(data.service)}`,
    78→    `*Booking Type:* ${data.bookingType === 'callout' ? 'Call-Out Service' : 'In-Sanctuary'}`,
    79→  ];
    80→  if (data.bookingType === 'callout') {
    81→    lines.push(`*Call-Out Zone:* ${getCalloutLabel(data.calloutZone)}`);
    82→    if (data.calloutAddress) lines.push(`*Address:* ${data.calloutAddress}`);
    83→    const svcPrice = servicePrices[data.service] || 0;
    84→    const zone = calloutZones.find(z => z.id === data.calloutZone);
    85→    const fee = zone?.fee || 0;
    86→    if (fee > 0) lines.push(`*Total:* K${svcPrice} + K${fee} call-out = K${svcPrice + fee}`);
    87→    else lines.push(`*Total:* K${svcPrice} + call-out fee (TBD)`);
    88→  }
    89→  lines.push(`*Therapist:* ${getTherapistLabel(data.therapist)}`, `*Payment:* ${getPaymentLabel(data.paymentMethod)}`);
    90→  if (data.message) lines.push(`*Notes:* ${data.message}`);
    91→  lines.push('', 'Please confirm my booking. Thank you!');
    92→  return encodeURIComponent(lines.join('\n'));
    93→}
