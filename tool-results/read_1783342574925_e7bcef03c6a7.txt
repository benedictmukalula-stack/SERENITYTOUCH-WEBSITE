    21→const formatDate = (dateStr: string) => {
    22→  try {
    23→    const d = new Date(dateStr);
    24→    return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    25→  } catch {
    26→    return dateStr;
    27→  }
    28→};
    29→
    30→const timeAgo = (dateStr: string) => {
    31→  try {
    32→    const now = Date.now();
    33→    const then = new Date(dateStr).getTime();
    34→    const diff = Math.floor((now - then) / 1000);
    35→    if (diff < 60) return `${diff}s ago`;
    36→    if (diff < 3600) return `${Math.floor(diff / 60)} min ago`;
    37→    if (diff < 86400) return `${Math.floor(diff / 3600)} hour ago`;
    38→    return `${Math.floor(diff / 86400)} days ago`;
    39→  } catch {
    40→    return '';
    41→  }
    42→};
