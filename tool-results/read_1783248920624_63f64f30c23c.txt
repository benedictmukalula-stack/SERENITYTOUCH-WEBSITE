    70→function getServiceLabel(val: string) { return serviceOptions.find(s => s.value === val)?.label || val; }
    71→function getTherapistLabel(val: string) { return therapistOptions.find(t => t.value === val)?.label || val; }
    72→function getPaymentLabel(val: string) { return paymentMethods.find(p => p.id === val)?.label || val; }

    74→function getCalloutLabel(val: string) {
    75→  const z = calloutZones.find(z => z.id === val);
    76→  return z ? (z.fee > 0 ? `${z.label} (K${z.fee} call-out fee)` : `${z.label} (Custom quote)`) : val;
    77→}

    79→function buildWhatsAppMessage(data: typeof defaultFormData): string {
    80→  const lines = [
    81→    `*Serenity Touch Spa — Booking Request*`, ``,
    82→    `*Name:* ${data.name}`, `*Email:* ${data.email}`, `*Phone:* ${data.phone || 'Not provided'}`,
    83→    `*Date:* ${data.date}`, `*Service:* ${getServiceLabel(data.service)}`,
    84→    `*Booking Type:* ${data.bookingType === 'callout' ? 'Call-Out Service' : 'In-Spa'}`,
    85→  ];
    86→  if (data.bookingType === 'callout') {
    87→    lines.push(`*Call-Out Zone:* ${getCalloutLabel(data.calloutZone)}`);
    88→    if (data.calloutAddress) lines.push(`*Address:* ${data.calloutAddress}`);
    89→    const svcPrice = servicePrices[data.service] || 0;
    90→    const zone = calloutZones.find(z => z.id === data.calloutZone);
    91→    const fee = zone?.fee || 0;
    92→    if (fee > 0) lines.push(`*Total:* K${svcPrice} + K${fee} call-out = K${svcPrice + fee}`);
    93→    else lines.push(`*Total:* K${svcPrice} + call-out fee (TBD)`);
    94→  }
    95→  lines.push(`*Therapist:* ${getTherapistLabel(data.therapist)}`, `*Payment:* ${getPaymentLabel(data.paymentMethod)}`);
    96→  if (data.message) lines.push(`*Notes:* ${data.message}`);
    97→  lines.push('', 'Please confirm my booking. Thank you!');
    98→  return encodeURIComponent(lines.join('\n'));
    99→}
