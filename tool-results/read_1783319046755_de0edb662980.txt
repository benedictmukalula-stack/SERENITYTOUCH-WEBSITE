   111→   111→// Deterministic "booked" check for realistic slot availability
   112→   112→function isSlotBooked(dateStr: string, time: string): boolean {
   113→   113→  let hash = 0;
   114→   114→  const str = dateStr + time;
   115→   115→  for (let i = 0; i < str.length; i++) {
   116→   116→    hash = ((hash << 5) - hash) + str.charCodeAt(i);
   117→   117→    hash |= 0;
   118→   118→  }
   119→   119→  return Math.abs(hash) % 5 === 0;
   120→   120→}

   122→   122→function formatDateDisplay(dateStr: string): string {
   123→   123→  if (!dateStr) return '';
   124→   124→  const d = new Date(dateStr + 'T00:00:00');
   125→   125→  return d.toLocaleDateString('en-GB', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' });
   126→   126→}

   134→   134→function toDateString(year: number, month: number, day: number): string {

   142→   142→function isSameDay(dateStr: string, year: number, month: number, day: number): boolean {

   146→   146→function isToday(year: number, month: number, day: number): boolean {
   147→   147→  const today = new Date();

   178→   178→  if (data.message) lines.push(`*Notes:* ${data.message}`);
   179→   179→  lines.push('', 'Please confirm my booking. Thank you!');
   180→   180→  return encodeURIComponent(lines.join('\n'));
