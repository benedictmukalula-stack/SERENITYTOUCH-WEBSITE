   114→// Deterministic "booked" check for realistic slot availability
   115→function isSlotBooked(dateStr: string, time: string): boolean {
   116→  let hash = 0;
   117→  const str = dateStr + time;
   118→  for (let i = 0; i < str.length; i++) {
   119→    hash = ((hash << 5) - hash) + str.charCodeAt(i);
   120→    hash |= 0;
   121→  }
   122→  return Math.abs(hash) % 5 === 0;
   123→}

   125→function formatDateDisplay(dateStr: string): string {
   126→  if (!dateStr) return '';
   127→  const d = new Date(dateStr + 'T00:00:00');
   128→  return d.toLocaleDateString('en-GB', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' });
   129→}
   130→
   131→function formatCalendarMonth(year: number, month: number): string {
   132→  return new Date(year, month).toLocaleDateString('en-US', { month: 'long', year: 'numeric' });
   133→}
   134→
   135→function pad(n: number): string { return n < 10 ? `0${n}` : `${n}`; }
   136→
   137→function toDateString(year: number, month: number, day: number): string {
   138→  return `${year}-${pad(month + 1)}-${pad(day)}`;
   139→}
   140→
   141→function isSunday(year: number, month: number, day: number): boolean {
   142→  return new Date(year, month, day).getDay() === 0;
   143→}
   144→
   145→function isSameDay(dateStr: string, year: number, month: number, day: number): boolean {
   146→  return dateStr === toDateString(year, month, day);
   147→}
   148→
   149→function isToday(year: number, month: number, day: number): boolean {
   150→  const today = new Date();
   151→  return year === today.getFullYear() && month === today.getMonth() && day === today.getDate();
   152→}
   153→
   154→function isPastDate(year: number, month: number, day: number): boolean {
   155→  const today = new Date();
   156→  today.setHours(0, 0, 0, 0);
   157→  return new Date(year, month, day) < today;
   158→}

   162→function buildWhatsAppMessage(data: typeof defaultFormData): string {
   163→  const lines = [
   164→    `*Serenity Touch Spa — Booking Request*`, ``,
   165→    `*Name:* ${data.name}`, `*Email:* ${data.email}`, `*Phone:* ${data.phone || 'Not provided'}`,
   166→    `*Date:* ${data.date ? formatDateDisplay(data.date) : 'Not selected'}`,
   167→    `*Time:* ${data.time || 'Not selected'}`,
   168→    `*Service:* ${getServiceLabel(data.service)}`,
   169→    `*Booking Type:* ${data.bookingType === 'callout' ? 'Call-Out Service' : 'In-Spa'}`,
   170→  ];
   171→  if (data.bookingType === 'callout') {
   172→    lines.push(`*Call-Out Zone:* ${getCalloutLabel(data.calloutZone)}`);
   173→    if (data.calloutAddress) lines.push(`*Address:* ${data.calloutAddress}`);
   174→    const svcPrice = servicePrices[data.service] || 0;
   175→    const zone = calloutZones.find(z => z.id === data.calloutZone);
   176→    const fee = zone?.fee || 0;
   177→    if (fee > 0) lines.push(`*Total:* K${svcPrice} + K${fee} call-out = K${svcPrice + fee}`);
   178→    else lines.push(`*Total:* K${svcPrice} + call-out fee (TBD)`);
   179→  }
   180→  lines.push(`*Therapist:* ${getTherapistLabel(data.therapist)}`, `*Payment:* ${getPaymentLabel(data.paymentMethod)}`);
   181→  if (data.message) lines.push(`*Notes:* ${data.message}`);
   182→  lines.push('', 'Please confirm my booking. Thank you!');
   183→  return encodeURIComponent(lines.join('\n'));
   184→}
