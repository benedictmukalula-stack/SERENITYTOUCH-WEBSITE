   111→// Deterministic "booked" check for realistic slot availability
   112→function isSlotBooked(dateStr: string, time: string): boolean {
   113→  let hash = 0;
   114→  const str = dateStr + time;
   115→  for (let i = 0; i < str.length; i++) {
   116→    hash = ((hash << 5) - hash) + str.charCodeAt(i);
   117→    hash |= 0;
   118→  }
   119→  return Math.abs(hash) % 5 === 0;
   120→}

   122→function formatDateDisplay(dateStr: string): string {
   123→  if (!dateStr) return '';
   124→  const d = new Date(dateStr + 'T00:00:00');
   125→  return d.toLocaleDateString('en-GB', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' });
   126→}
   127→
   128→function formatCalendarMonth(year: number, month: number): string {
   129→  return new Date(year, month).toLocaleDateString('en-US', { month: 'long', year: 'numeric' });
   130→}
   131→
   132→function pad(n: number): string { return n < 10 ? `0${n}` : `${n}`; }
   133→
   134→function toDateString(year: number, month: number, day: number): string {
   135→  return `${year}-${pad(month + 1)}-${pad(day)}`;
   136→}
   137→
   138→function isSunday(year: number, month: number, day: number): boolean {
   139→  return new Date(year, month, day).getDay() === 0;
   140→}
   141→
   142→function isSameDay(dateStr: string, year: number, month: number, day: number): boolean {
   143→  return dateStr === toDateString(year, month, day);
   144→}
   145→
   146→function isToday(year: number, month: number, day: number): boolean {
   147→  const today = new Date();
   148→  return year === today.getFullYear() && month === today.getMonth() && day === today.getDate();
   149→}
   150→
   151→function isPastDate(year: number, month: number, day: number): boolean {
   152→  const today = new Date();
   153→  today.setHours(0, 0, 0, 0);
   154→  return new Date(year, month, day) < today;
   155→}

   159→function buildWhatsAppMessage(data: typeof defaultFormData): string {
   160→  const lines = [
   161→    `*Serenity Touch Spa — Booking Request*`, ``,
   162→    `*Name:* ${data.name}`, `*Email:* ${data.email}`, `*Phone:* ${data.phone || 'Not provided'}`,

   165→    `*Service:* ${getServiceLabel(data.service)}`,
   166→    `*Booking Type:* ${data.bookingType === 'callout' ? 'Call-Out Service' : 'In-Spa'}`,
   167→  ];
   168→  if (data.bookingType === 'callout') {
   169→    lines.push(`*Call-Out Zone:* ${getCalloutLabel(data.calloutZone)}`);
   170→    if (data.calloutAddress) lines.push(`*Address:* ${data.calloutAddress}`);
   171→    const svcPrice = servicePrices[data.service] || 0;
   172→    const zone = calloutZones.find(z => z.id === data.calloutZone);
   173→    const fee = zone?.fee || 0;
   174→    if (fee > 0) lines.push(`*Total:* K${svcPrice} + K${fee} call-out = K${svcPrice + fee}`);
   175→    else lines.push(`*Total:* K${svcPrice} + call-out fee (TBD)`);
   176→  }
   177→  lines.push(`*Therapist:* ${getTherapistLabel(data.therapist)}`, `*Payment:* ${getPaymentLabel(data.paymentMethod)}`);
   178→  if (data.message) lines.push(`*Notes:* ${data.message}`);
   179→  lines.push('', 'Please confirm my booking. Thank you!');
   180→  return encodeURIComponent(lines.join('\n'));
   181→}
